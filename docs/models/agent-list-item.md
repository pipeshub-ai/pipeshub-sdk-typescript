# AgentListItem

Agent projection returned by `GET /agents`.

This is the list-view envelope item emitted by the Python backend and
forwarded by the Node gateway. It is not the full detail projection
used by `GET /agents/{agentKey}`.


## Example Usage

```typescript
import { AgentListItem } from "@pipeshub-ai/sdk/models";

let value: AgentListItem = {
  id: "agentInstances/e6f848ca-e2ab-4594-9925-e1136629f474",
  key: "e6f848ca-e2ab-4594-9925-e1136629f474",
  rev: "_lkNlcOm---",
  createdAtTimestamp: 962996,
  createdBy: "<value>",
  isActive: false,
  isDeleted: true,
  isServiceAccount: false,
  models: [
    "<value 1>",
    "<value 2>",
  ],
  name: "<value>",
  tags: [
    "<value 1>",
  ],
  updatedAtTimestamp: 536230,
  shareWithOrg: false,
  toolsets: [
    {},
  ],
  knowledge: [],
  canView: false,
  canShare: true,
  canEdit: true,
  canDelete: true,
  userRole: "OWNER",
  accessType: "INDIVIDUAL",
};
```
