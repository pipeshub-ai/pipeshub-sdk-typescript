# ConnectorInstance

A connector instance: its type's registry entry plus the instance's own state.

## Example Usage

```typescript
import { ConnectorInstance } from "@pipeshub-ai/sdk/models";

let value: ConnectorInstance = {
  key: "7b387394-1248-4fb6-810d-e8ac9ac226eb",
  name: "Company Google Drive",
  type: "Drive Workspace",
  documentationLinks: [
    {
      title: "Azure AD App Registration Setup",
      type: "setup",
    },
  ],
  authType: "OAUTH",
  scope: "team",
  status: "IDLE",
  config: {
    documentationLinks: [
      {
        title: "Azure AD App Registration Setup",
        type: "setup",
      },
    ],
    sync: {
      scheduledConfig: {
        timezone: "UTC",
      },
      realtimeConfig: {
        connectionType: "WEBSOCKET",
      },
    },
  },
};
```
