# ConnectorRegistryEntry

A connector type the org can add.

## Example Usage

```typescript
import { ConnectorRegistryEntry } from "@pipeshub-ai/sdk/models";

let value: ConnectorRegistryEntry = {
  name: "OneDrive",
  type: "OneDrive",
  appGroup: "Microsoft 365",
  appCategories: [
    "Storage",
  ],
  iconPath: "/icons/connectors/onedrive.svg",
  documentationLinks: [
    {
      title: "Azure AD App Registration Setup",
      type: "setup",
    },
  ],
  personalConnectorType: "Notion Personal",
  scope: "team",
};
```
